.formulation[]?.properties[]? | select(.name == "monolith:workflow:commit" or .name == "monolith:workflow:run_id")   # the same commit and run id of the producing CI run

.formulation[]?.components[]? | select(.name == "versions.lock")   # hashes configs/portage/versions.lock, which this repository does not have until B01
.formulation[]?.components[]? | select(.name | test("^linux-.*[.]tar[.]xz$"))   # reads the kernel ebuild's Manifest (configs/overlay/sys-kernel/monolith-kernel/Manifest), which this repository does not have yet
.dependencies[]? | select(.ref == "themonolith-0.1.0-rc1")   # the product's dependsOn roots come from configs/portage/world, which this repository does not have until B01
